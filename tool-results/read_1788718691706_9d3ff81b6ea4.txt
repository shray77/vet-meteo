   293→    const ty=y(72);
   294→    thr='<line x1="'+P+'" y1="'+ty.toFixed(1)+'" x2="'+(W-P)+'" y2="'+ty.toFixed(1)+'" stroke="var(--red)" stroke-width="0.7" stroke-dasharray="3 2" opacity="0.6"/>';
   295→  }
   296→  let line='';
   297→  if(n>=2){
   298→    const pts=vs.map((v,i)=>x(i).toFixed(1)+','+y(v).toFixed(1)).join(' ');
   299→    line='<polyline points="'+pts+'" fill="none" stroke="'+ZFG[z]+'" stroke-width="1.4" stroke-linejoin="round"/>';
   300→  }
   301→  return '<svg class="spark" width="'+W+'" height="'+H+'" viewBox="0 0 '+W+' '+H+'">'
   302→    +'<title>THI: '+fmt(vs[0])+' → '+fmt(vs[n-1])+' · '+n+' срез(а/ов)</title>'
   303→    +line+thr
   304→    +'<circle cx="'+x(n-1).toFixed(1)+'" cy="'+y(vs[n-1]).toFixed(1)+'" r="2.2" fill="'+ZFG[z]+'"/>'
   305→    +'</svg>';
   306→}

   357→function renderMap(){
   358→  const svg=$('mapSvg'),L=S.latest,G=S.geo;
   359→  if(!L||!G){svg.innerHTML='';return;}
   360→  const bb=G.bbox,W=400,PAD=26;
   361→  const spanLon=bb.maxLon-bb.minLon,spanLat=bb.maxLat-bb.minLat;
   362→  const H=Math.max(160,Math.round(PAD*2+spanLat/spanLon*(W-2*PAD)));
   363→  svg.setAttribute('viewBox','0 0 '+W+' '+H);
   364→  const px=lon=>PAD+(lon-bb.minLon)/spanLon*(W-2*PAD);
   365→  const py=lat=>PAD+(bb.maxLat-lat)/spanLat*(H-2*PAD);
   366→  let g='';
   367→  for(let lon=Math.ceil(bb.minLon);lon<=bb.maxLon;lon++)
   368→    g+='<line x1="'+px(lon)+'" y1="'+(PAD-6)+'" x2="'+px(lon)+'" y2="'+(H-PAD+6)+'" stroke="var(--border2)" stroke-width="0.6"/><text x="'+px(lon)+'" y="'+(H-6)+'" font-size="7" fill="var(--dim)" text-anchor="middle">'+lon+'°E</text>';
   369→  for(let lat=Math.ceil(bb.minLat);lat<=bb.maxLat;lat++)
   370→    g+='<line x1="'+(PAD-6)+'" y1="'+py(lat)+'" x2="'+(W-PAD+6)+'" y2="'+py(lat)+'" stroke="var(--border2)" stroke-width="0.6"/><text x="8" y="'+(py(lat)+2.5)+'" font-size="7" fill="var(--dim)">'+lat+'°N</text>';
   371→  g+='<rect x="'+PAD+'" y="'+PAD+'" width="'+(W-2*PAD)+'" height="'+(H-2*PAD)+'" fill="none" stroke="var(--border)" stroke-width="1"/>';
   372→  const st=L.stations||[];
   373→  const top=[...st].sort((a,b)=>b.thi-a.thi).slice(0,3);
   374→  for(const s of st){
   375→    const z=zone(s.thi),x=px(s.lon),y=py(s.lat);
   376→    const isTop=top.find(t=>t.id===s.id);
   377→    if(isTop)g+='<circle cx="'+x+'" cy="'+y+'" r="11" fill="none" stroke="'+ZFG[z]+'" stroke-width="1" opacity="0.5"/>';
   378→    g+='<circle cx="'+x+'" cy="'+y+'" r="5" fill="'+ZFG[z]+'" opacity="0.9" stroke="#0b0d08" stroke-width="1"><title>'+s.name+' · T='+fmt(s.t)+'°C · THI='+fmt(s.thi)+' · BRD='+s.brd+'</title></circle>';
   379→    if(isTop)g+='<text x="'+x+'" y="'+(y-10)+'" font-size="8" fill="var(--muted)" text-anchor="middle">'+s.name.toUpperCase()+'</text>';
   380→  }
   381→  g+='<text x="'+(W-PAD)+'" y="'+(PAD-12)+'" font-size="8" fill="var(--accent)" text-anchor="end">THI MAX '+fmt(Math.max(...st.map(s=>s.thi)))+'</text>';
   382→  svg.innerHTML=g;
   383→  $('mapRight').textContent='срез '+hhmm(L.generatedAt);
   384→}

   387→function renderTable(){
   388→  const L=S.latest,tb=$('stTable').tBodies[0];
   389→  if(!L){tb.innerHTML='<tr><td colspan="9" class="small-dim">нет данных</td></tr>';return;}

   391→    const k=S.sort.k,va=a[k],vb=b[k];
   392→    if(typeof va==='string')return S.sort.dir*va.localeCompare(vb);
   393→    return S.sort.dir*((va==null?-(1/0):va)-(vb==null?-(1/0):vb));
   394→  });
   395→  tb.innerHTML=st.map(s=>{
   396→    const z=zone(s.thi),zt=zone(s.thiMaxToday);
   397→    const bcol=s.brd==null?'var(--muted)':s.brd>=75?'var(--z4fg)':s.brd>=50?'var(--z3fg)':s.brd>=25?'var(--z2fg)':'var(--z1fg)';

   400→      +'<td>'+fmt(s.t)+'</td>'
   401→      +'<td>'+fmt(s.rh,0)+'</td>'
   402→      +'<td class="small-dim">'+fmt(s.wind)+' м/с '+wdir2c(s.wdir)+'</td>'

   405→      +'<td><span class="chip z'+zt+'" style="min-width:44px;font-size:11px">'+fmt(s.thiMaxToday)+'</span></td>'
   406→      +'<td style="color:'+bcol+';font-weight:700">'+(s.brd==null?'—':fmt(s.brd,0))+'</td>'

   414→}

   418→    if(S.sort.k===k)S.sort.dir*=-1; else S.sort={k,dir:-1};
   419→    renderTable();
